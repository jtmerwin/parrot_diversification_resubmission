# 1. Read phylogeny #
T <- readTrees("/media/pc/newhardrive/programs/revbayes-v1.2.1/parrots/trimmedParrotTreeForBrian.3.3.2023.matchedMusscrat.tre")[1]
ntips <- T.ntips()
nbranches <- 2 * ntips - 2

###morph PC1 and Hue ###
# 2. Read matrix of morph PC1 and Hue continuous traits #
#cont <- readContinuousCharacterData("/media/pc/newhardrive/programs/revbayes-v1.2.1/parrots/Parrots_BS_PCA1.nex")
cont <- readContinuousCharacterData("/media/pc/newhardrive/programs/revbayes-v1.2.1/parrots/Parrots_HUE.nex")


# 3. Read vector of habitat and continent #
#disc <- readDiscreteCharacterData("/media/pc/newhardrive/programs/revbayes-v1.2.1/parrots/parrot_continents.nex")
disc <- readDiscreteCharacterData("/media/pc/newhardrive/programs/revbayes-v1.2.1/parrots/parrot_habitat-23-may-2024.nex")

num_disc_states <- disc.getStateDescriptions().size()

# 4. Initialize vectors of moves and monitors for MCMC #
moves    = VectorMoves()
monitors = VectorMonitors()

# 5. Specify phylogenetic model (take imported phylogeny) #
tree <- T

# 6. Specify model of discrete character evolution #
Q <- fnJC(num_disc_states)
lambda ~ dnLoguniform(1e-4, 1)
moves.append( mvScale(lambda, weight=1.0) )
X ~ dnPhyloCTMCDASiteIID(tree, Q, branchRates=lambda, nSites=1, type="Standard")
X.clamp(disc)
moves.append( mvCharacterHistory(ctmc=X, qmap_site=Q, graph="node",   proposal="rejection", weight=20.0) )
moves.append( mvCharacterHistory(ctmc=X, qmap_site=Q, graph="branch", proposal="rejection", weight=20.0) )
for(i in 1:nbranches) {
    num_changes[i] := sum(X.numCharacterChanges(i))
}
total_num_changes := sum(num_changes)

# 7 Specifiy the model of continuous character evolution #
beta_root ~ dnLoguniform(1e-3, 1)
moves.append( mvScale(beta_root, weight=1.0) )
#evaluate
expected_number_of_shifts <- 25
rate_shift_probability    <- expected_number_of_shifts / nbranches
sd = 0.578
rate_shift_distribution = dnLognormal(-sd^2/2, sd)
for(i in nbranches:1) {
    branch_rate_multiplier[i] ~ dnReversibleJumpMixture(1, rate_shift_distribution, Probability(1 - rate_shift_probability) )
    if ( tree.isRoot( tree.parent(i) ) ) {
       background_rates[i] := beta_root * branch_rate_multiplier[i]
    } else {
       background_rates[i] := background_rates[tree.parent(i)] * branch_rate_multiplier[i]
    }
    branch_rate_shift[i] := ifelse( branch_rate_multiplier[i] == 1, 0, 1 )
    moves.append( mvRJSwitch(branch_rate_multiplier[i], weight=1) )
    moves.append( mvScale(branch_rate_multiplier[i], weight=1) )
}
num_rate_changes := sum( branch_rate_shift )

# use reversible jump mixture to sample from both state-dependent and state-independent models #
concentration <- 1.0
proportional_zeta ~ dnReversibleJumpMixture( simplex(rep(1,num_disc_states)), dnDirichlet( rep(concentration, num_disc_states) ), p=0.5 )
moves.append( mvRJSwitch(proportional_zeta, weight=10.0) )
moves.append( mvBetaSimplex(proportional_zeta, weight=1.0) )
is_state_dependent := ifelse( proportional_zeta == simplex(rep(1,num_disc_states)), 0.0, 1.0)
zeta := proportional_zeta * num_disc_states
for(i in 1:nbranches) {
    state_branch_rate[i] := sum(X.relativeTimeInStates(i,1) * abs(zeta))
}
branch_rates := state_branch_rate * background_rates
Y ~ dnPhyloBrownianREML(tree, branchRates=branch_rates^0.5)
Y.clamp(cont)
mymodel = model(zeta)

# 8. Setup MCMC #
#monitors.append( mnModel(filename="/media/pc/newhardrive/programs/revbayes-v1.2.1/MuSSCRat_parrots_PCA1BS_continent_5RS.log", printgen=10) )  	### CHANGE FILE NAME
#monitors.append( mnModel(filename="/media/pc/newhardrive/programs/revbayes-v1.2.1/MuSSCRat_parrots_Hue_continent_5RS-test.log", printgen=10) )  	### CHANGE FILE NAME
#monitors.append( mnModel(filename="/media/pc/newhardrive/programs/revbayes-v1.2.1/parrots/MuSSCRat_parrots_PCA1BS_continent_5RS.log", printgen=10) )  
monitors.append( mnModel(filename="/media/pc/newhardrive/programs/revbayes-v1.2.1/re-do/parrots/MuSSCRat_parrots_Hue_habitat_25RS.log", printgen=10) )  


monitors.append( mnScreen(printgen=1000, zeta, total_num_changes, is_state_dependent) )
mymcmc = mcmc(mymodel, monitors, moves, nruns=1, combine="mixed")
mymcmc.run(generations=1000000)
